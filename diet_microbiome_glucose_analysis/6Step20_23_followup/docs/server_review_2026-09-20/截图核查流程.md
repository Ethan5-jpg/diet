## 第二步：三个模型全局FDR共同通过集合

第一步已收到2026-09-20 13:06:11截图，转录和解释见同目录`Step20_指标重复与核心菌覆盖解读.md`。以下命令按完整路径键配对，读取既有全局q列。`joint3`表示三模型均q<0.05；`not_joint3`包括至少一个模型未通过或q缺失的路径，缺失计数单独显示。

```bash
python3 - <<'PY'
from pathlib import Path
import pandas as pd

run = Path("outputs/followup_20260920T044755_767120Z")
t = run / "tables"
x = pd.read_csv(t / "21_all1490_three_model_results.csv")
flag = x.frozen_retained_both.astype(str).str.strip().str.lower()
assert flag.isin(["true", "false"]).all(), "Unexpected frozen flag"
x = x[flag.eq("true")].copy()
keys = ["exposure", "outcome_field", "species"]
models = ["primary", "strict", "protocol"]
assert set(x.model) == set(models), "Model set mismatch"
q = x.pivot(index=keys, columns="model", values="FDR_BH_global_all_mediation")
q = q.reindex(columns=models).apply(pd.to_numeric, errors="raise")
assert len(x) == 2613 and q.shape == (871, 3), "Frozen path count mismatch"
p = q.lt(0.05)
p["joint3"] = p.all(axis=1)
print("RUN:", run.name)
print("Missing q:", q.isna().sum().to_dict())
print("Passed among 871:", p.sum().to_dict())

m = pd.read_csv(t / "22_species_evidence_matrix_to_complete.csv")
m = m[["species", "taxon_id"]].drop_duplicates()
d = p.reset_index().merge(m, on="species", how="left", validate="many_to_one")
assert d.taxon_id.notna().all(), "Taxon mapping incomplete"
d = d[d.taxon_id.isin(["T023", "T046", "T068", "T096"])]
g = d.groupby("taxon_id")
s = g[models + ["joint3"]].sum().astype(int)
s.insert(0, "frozen", g.size())
s["not_joint3"] = s.frozen - s.joint3
print("\n=== Core taxa: existing global q < 0.05 ===")
print(s.to_string())

```

## 第三步：定位T046的3条路径并检查领域覆盖

第二步已收到2026-09-20 13:13:09截图：joint3=832；四株合计87/90共同通过，T046的3条路径仅strict全局q未达到0.05。以下输出约二十行，L=水平、V=波动、R=风险/范围。

```bash
python3 - <<'PY'
from pathlib import Path
import pandas as pd

run = Path("outputs/followup_20260920T044755_767120Z")
t = run / "tables"
x = pd.read_csv(t / "21_all1490_three_model_results.csv")
flag = x.frozen_retained_both.astype(str).str.strip().str.lower()
assert flag.isin(["true", "false"]).all()
x = x[flag.eq("true")].copy()
models = ["primary", "strict", "protocol"]
assert set(x.model) == set(models)
q = x.pivot(index=["exposure", "outcome_field", "species"],
            columns="model", values="FDR_BH_global_all_mediation")
q = q.reindex(columns=models).apply(pd.to_numeric, errors="raise")
assert len(x) == 2613 and q.shape == (871, 3)
assert q.notna().all().all(), "Unexpected missing q"
q["joint3"] = q.lt(0.05).all(axis=1)
m = pd.read_csv(t / "22_species_evidence_matrix_to_complete.csv")
d = q.reset_index().merge(m[["species", "taxon_id"]].drop_duplicates(),
                         on="species", how="left", validate="many_to_one")
f = pd.read_csv(t / "20_dictionary_used.csv").set_index("field").domain
d["domain"] = d.outcome_field.map(f).map({
    "glycemic_level": "L", "glycemic_variability": "V",
    "glycemic_risk_range": "R"})
assert d[["taxon_id", "domain"]].notna().all().all()
bad = d[d.taxon_id.eq("T046") & ~d.joint3]
assert len(bad) == 3
print("RUN:", run.name)
print("\n=== T046: paths not passing all 3 global q thresholds ===")
print(bad[["exposure", "outcome_field", "domain"] + models]
      .sort_values(["exposure", "outcome_field"])
      .to_string(index=False, float_format=lambda v: f"{v:.8g}"))
print("\n=== All frozen paths: L=level V=variability R=risk/range ===")
print(d.groupby("domain").joint3.agg(["size", "sum"])
      .rename(columns={"size": "frozen", "sum": "joint3"}).to_string())
ids = ["T023", "T046", "T068", "T096"]
c = d[d.taxon_id.isin(ids) & d.joint3]
b = pd.crosstab(c.taxon_id, c.domain).reindex(
    index=ids, columns=["L", "V", "R"], fill_value=0)
b["domains"] = b.gt(0).sum(axis=1)
print("\n=== Core joint3 paths by domain ===")
print(b.to_string())
PY
```
