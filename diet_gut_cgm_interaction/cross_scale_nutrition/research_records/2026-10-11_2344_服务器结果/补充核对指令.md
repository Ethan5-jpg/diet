import hashlib
import json
from pathlib import Path
import pandas as pd

def need(ok, msg):
    if not ok:
        raise SystemExit("CHECK STOPPED: " + msg)

expected_kind = "SERVER_INPUTS"
matches = list(Path("outputs").glob("common_20261011T154438_631520Z_*"))
need(len(matches) == 1, "Expected exactly one run with the screenshot timestamp")
run = matches[0]
manifest_bytes = (run / "manifest.json").read_bytes()
m = json.loads(manifest_bytes)
need(m["data_kind"] == expected_kind, "Wrong real/synthetic mode")
need(m["version"] == "2026-10-11.cross.1", "Wrong analysis version")
need(m["status"] == "COMPLETE" and m["run_id"] == run.name, "Incomplete or mismatched run")
files = ["tables/10_interactions.csv", "tables/11_comparisons.csv", "tables/14_conditional_slopes.csv"]
def verify():
    need((run / "manifest.json").read_bytes() == manifest_bytes, "Manifest changed")
    for name in files:
        digest = hashlib.sha256((run / name).read_bytes()).hexdigest()
        need(digest == m["output_sha256"][name], "Saved table changed: " + name)
verify()
r, c, s = [pd.read_csv(run / name) for name in files]
need(len(r) == 44 and len(c) == 40 and len(s) == 176, "Unexpected registry size")
need(r.status.eq("ESTIMATED").all() and c.status.eq("ESTIMATED").all(), "Unestimated rows")

def pick(frame, **keys):
    mask = pd.Series(True, index=frame.index)
    for key, value in keys.items():
        mask &= frame[key].eq(value)
    rows = frame.loc[mask]
    need(len(rows) == 1, "Missing or duplicate row: " + str(keys))
    return rows.iloc[0]

def f(x):
    return "NA" if pd.isna(x) else "{:.5g}".format(float(x))

def ci(row):
    return "{} [{},{}]".format(f(row.beta), f(row.low), f(row.high))

def diff_line(code, row):
    return "{} {} {} {} {} {} {}".format(code, row.scale, row.outcome, row.taxon,
                                         ci(row), f(row.q_family), f(row.q_all40))

def est_line(code, variant, scale, outcome, taxon):
    row = pick(r, variant=variant, scale=scale, outcome=outcome, taxon=taxon)
    return "{} {} {} {} {} {} {} {} {}".format(code, scale, outcome, taxon,
        int(row.n_people), int(row.n_days), ci(row), f(row.q_family), f(row.q_all44))

first = ["CHECK 1/3 | SAVED CONTRASTS | " + expected_kind + " | hashes verified",
         "X=W-B J=joint-single D=classified-total S=N0-F A=N1-N0 C=MeanCond-base",
         "All SD contrasts + all four scale contrasts; qF=saved family; q40=all 40.",
         "Code Sc Y Taxon Difference [L95,U95] qF q40"]
labels = [("X", "W_minus_B"), ("J", "joint_minus_single"),

          ("A", "N1_minus_N0"), ("C", "conditional_minus_base")]

targets = [("B", "SD", "T068"), ("W", "SD", "T096"), ("W", "Mean", "T096")]
second = ["CHECK 2/3 | DENOMINATOR / NUTRITION | " + expected_kind,
          "D=classified denominator; N0/N1=identical nutrition sample; qF=saved family.",
          "Code Sc Y Taxon People Days Beta [L95,U95] qF q44"]
for scale in ["B", "W"]:

        for taxon in ["T068", "T096"]:
            second.append(est_line("D", "F_classified_joint", scale, outcome, taxon))
for scale, outcome, taxon in targets:
    for variant in ["N0_joint", "N1_joint"]:
        second.append(est_line(variant[:2], variant, scale, outcome, taxon))
second += ["Prior observed W Mean T096: saved direct differences", "Code Sc Y Taxon Difference [L95,U95] qF q40"]
for code, label in labels[1:5]:
    second.append(diff_line(code, pick(c, label=label, scale="W", outcome="Mean", taxon="T096")))

third = ["CHECK 3/3 | CONDITIONAL SD / PRIMARY DIET SLOPES | " + expected_kind,
         "C=SD given Mean + Mean^2 and microbe products; qF=4, q44=all interactions.",
         "Code Sc Y Taxon People Days Beta [L95,U95] qF q44"]
for scale in ["B", "W"]:
    for taxon in ["T068", "T096"]:
        third.append(est_line("C", "F_mean_adjusted_joint", scale, "SD", taxon))
third += ["F_joint diet slopes: mg/dL per NOVA4 +10pp; other microbe at F mean.",
          "Q25/Q75 fixed F anchors; pointwise CI; dIQR=Q75-minus-Q25 slope difference.",
          "Sc Y Taxon Point MicrobeCLRZ Slope_or_difference [L95,U95]"]
for scale, outcome, taxon in targets:
    for point in ["q25", "q75", "q75_minus_q25"]:

        label = "dIQR" if point == "q75_minus_q25" else point
        third.append("{} {} {} {} {} {}".format(scale, outcome, taxon, label, f(row.microbe), ci(row)))

          "Read after prior results; this is not independent validation.", "RESULT_DIR=" + str(run.resolve())]
